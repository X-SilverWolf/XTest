if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    cmake_minimum_required(VERSION 3.28)
    project(XTest VERSION 1.0.0 LANGUAGES C CXX)
    include("${CMAKE_CURRENT_LIST_DIR}/cmake/Standalone.cmake")
    x_enable_business_releases()

    option(XTEST_BUILD_TBASIC "Build the window test application" ON)
    option(XTEST_BUILD_TRUNTIME "Build the runtime test application" ON)
    option(XTEST_BUILD_PYTHON "Build the Python extension" OFF)
    option(XTEST_BUILD_GUI "Build the Qt Widgets test tool" OFF)
    option(XTEST_BUILD_IMGALGO "Expose the image algorithm headers" OFF)
    option(XTEST_BUILD_CUDA_DEMO "Build the CUDA runtime and OpenGL demo" OFF)

    if(XTEST_BUILD_TBASIC)
        include("${CMAKE_CURRENT_LIST_DIR}/TBasic/cmake/Dependencies.cmake")
    endif()
    if(XTEST_BUILD_TRUNTIME)
        include("${CMAKE_CURRENT_LIST_DIR}/TRunTime/cmake/Dependencies.cmake")
    endif()
    if(XTEST_BUILD_CUDA_DEMO)
        include("${CMAKE_CURRENT_LIST_DIR}/src/Tests/cmake/Dependencies.cmake")
    endif()
    if(XTEST_BUILD_PYTHON)
        include("${CMAKE_CURRENT_LIST_DIR}/src/PyTools/PyTest/cmake/Dependencies.cmake")
    endif()
    if(XTEST_BUILD_IMGALGO)
        include("${CMAKE_CURRENT_LIST_DIR}/src/ImgAlgo/cmake/Dependencies.cmake")
    endif()
endif()

if(TARGET X::XGui AND (NOT DEFINED XTEST_BUILD_TBASIC OR XTEST_BUILD_TBASIC))
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/TBasic")
endif()
if(TARGET X::XRunTime AND (NOT DEFINED XTEST_BUILD_TRUNTIME OR XTEST_BUILD_TRUNTIME))
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/TRunTime")
endif()
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/src/LibTest")
if(XTEST_BUILD_PYTHON)
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/src/PyTools/PyTest")
endif()
if(XTEST_BUILD_GUI)
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/src/UITools/TestTool")
endif()
if(XTEST_BUILD_IMGALGO)
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/src/ImgAlgo")
endif()
if(XTEST_BUILD_CUDA_DEMO)
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/src/Tests")
endif()
